cmake_minimum_required(VERSION 3.22.1)
project(sightline_lan_host LANGUAGES CXX)
add_library(node SHARED IMPORTED)
if(NOT NODE_MOBILE_SDK)
  message(FATAL_ERROR "Verified external Node Mobile SDK required")
endif()
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_MOBILE_SDK}/bin/${ANDROID_ABI}/libnode.so")
add_library(sightline_host SHARED lan_host.cpp)
target_include_directories(sightline_host PRIVATE "${NODE_MOBILE_SDK}/include/node")
target_compile_features(sightline_host PRIVATE cxx_std_20)
target_link_libraries(sightline_host PRIVATE node log)
